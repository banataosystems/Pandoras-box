test("update validates, binds schema fingerprint, then writes exactly one YAML file", async () => {
  const calls = [];
  const fetchFn = async (url, init) => {
    calls.push({ url: String(url), init });
    if (String(url).includes("/listPartitionedFileNames")) {
      return jsonResponse({
        success: true,
        reason: null,
        value: {
          versionInfo: {
            partitionerVersion: 13,
            projectSchemaFingerprint: "fingerprint-13",
          },
          fileNames: ["app-state"],
        },
      });
    }
    if (String(url).includes("/validateProjectYaml")) return jsonResponse({ success: true, reason: null, value: "" });
    if (String(url).includes("/updateProjectByYaml")) return jsonResponse({ success: true, reason: null, value: "" });
    throw new Error(`unexpected URL ${url}`);
  };

  const result = await executeFlutterFlowTool(
    "flutterflow.update-project-yaml",
    {
      accountId: "pandora-mobile",
      projectId: "pandoras-box-gj9hnb",
      fileKey: "app-state",
      fileContent: "fields: []\n",
      expectedSchemaFingerprint: "fingerprint-13",
    },
    configuration(),
    fetchFn,
  );

  assert.equal(result.updated, true);
  assert.equal(result.validated, true);
  assert.equal(calls.length, 3);
  assert.match(calls[0].url, /listPartitionedFileNames/);
  assert.match(calls[1].url, /validateProjectYaml/);
  assert.match(calls[2].url, /updateProjectByYaml/);
  assert.deepEqual(JSON.parse(calls[2].init.body), {
    projectId: "pandoras-box-gj9hnb",
    fileKeyToContent: { "app-state": "fields: []\n" },
  });
});
